# =========================================================
# MAKE SHORTCUTS  (run from project root)
# =========================================================

  make all                    → Run ALL sims sequentially (no GTKWave)
  make fw                     → Build firmware only
  make uart                   → UART loopback TB
  make pico                   → PicoRV32 TB
  make riscv_axi              → RISC-V AXI-Lite integration TB
  make axi_uart               → AXI-Lite UART TB
  make axi_sa                 → AXI-Lite standalone TB
  make soc_pico_uart_mem      → PicoRV32 mem-mapped UART SoC
  make soc                    → Full PicoRV32 + UART + AXI SoC
  make debug_soc              → Full SoC structured verification log
  make wave                   → Open last tb_top waveform in GTKWave
  make synth                  → Run Yosys synthesis
  make clean                  → Clean all build artifacts

  -Wno-PROCASSINIT \
  --top tb_uart_top \
  -Mdir obj_uart \
  -o sim_uart \

  rtl/uart_rx.v \
  rtl/uart_top.v \
  tb/tb_uart_top.v
